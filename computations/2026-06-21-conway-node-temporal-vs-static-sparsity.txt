STATIC AGGREGATE CONNECTIVITY VS TEMPORAL TRANSPORT

chronology permits 0->3
  contacts=[(1, 0, 1), (2, 1, 2), (3, 2, 3)]
  aggregate_edges=[(0, 1), (1, 2), (2, 3)]
  static reachable from 0=[0, 1, 2, 3]
  temporal reachable from 0=[0, 1, 2, 3] with earliest={0: -1, 1: 1, 2: 2, 3: 3}

same aggregate, chronology blocks 0->3
  contacts=[(1, 2, 3), (2, 1, 2), (3, 0, 1)]
  aggregate_edges=[(0, 1), (1, 2), (2, 3)]
  static reachable from 0=[0, 1, 2, 3]
  temporal reachable from 0=[0, 1] with earliest={0: -1, 1: 3}

same aggregate, partial block
  contacts=[(1, 1, 2), (2, 0, 1), (3, 2, 3)]
  aggregate_edges=[(0, 1), (1, 2), (2, 3)]
  static reachable from 0=[0, 1, 2, 3]
  temporal reachable from 0=[0, 1] with earliest={0: -1, 1: 2}
